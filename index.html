Code Example

Multiples of 3 or 5

This solution uses arithmetic-series sums and the inclusion-exclusion principle to avoid iteration. 

function solution(number) {
                if (number <= 0) return 0;

                const multiplesOfThreeCount = Math.floor((number - 1) / 3);
                const multiplesOfFiveCount = Math.floor((number - 1) / 5);
                const multiplesOfFifteenCount = Math.floor((number - 1) / 15);

                const sumOfMultiplesOfThree =
                    (3 * multiplesOfThreeCount * (multiplesOfThreeCount + 1)) / 2;

                const sumOfMultiplesOfFive =
                    (5 * multiplesOfFiveCount * (multiplesOfFiveCount + 1)) / 2;

                const sumOfMultiplesOfFifteen =
                    (15 * multiplesOfFifteenCount * (multiplesOfFifteenCount + 1)) / 2;

                return (
                    sumOfMultiplesOfThree +
                    sumOfMultiplesOfFive -
                    sumOfMultiplesOfFifteen
                );
            }
